--[[
    ════════════════════════════════════════════════════════════════
      AUTOFARM — URBAN RELOAD (LOCK ON SIGHT + AUTOFIRE)
    ════════════════════════════════════════════════════════════════
      LocalScript à placer dans  StarterPlayerScripts

      Comportement :
      1. SCAN — trouve les PNJ (Humanoid, Health > 0, non-joueurs,
         double vérification Players:GetPlayerFromCharacter).
      2. LOCK — prend le PNJ le plus proche avec ligne de vue dégagée
         (raycast caméra → tête). LockFov = 180 : il verrouille à
         360° autour de toi, sans avoir besoin de le regarder.
      3. AIM  — la caméra se tourne TOUTE SEULE sur la tête (lead sur
         la vélocité + micro-jitter) et tire. Le perso ne bouge jamais.
      4. FIRE — farm ON : tout part tout seul, AUCUNE touche à presser.
         Clic droit maintenu en permanence + clic gauche toutes les
         75 ms + Tool:Activate(). OFF ou RESET : tout est relâché.
      5. LOOT — spamme E en continu via VirtualInputManager pour
         ramasser tout ce qui passe à portée, sans bouger. Le scan
         ProximityPrompt ne sert plus qu'au compteur "Loots".
         WalkToLoot = true pour marcher au loot.

      ── POURQUOI PAS LE SILENTAIM ─────────────────────────────────
      Le redirect TargetFilter de SilentAimNPC.lua suppose que l'arme
      lise Mouse.Hit.p côté client. Urban Reload ne suit pas ce modèle :
      ici on pilote directement la caméra + les entrées, ce qui marche
      quel que soit le système de tir du jeu.
    ════════════════════════════════════════════════════════════════
]]

-- ═══════════════════════ SERVICES ═══════════════════════

local Players             = game:GetService("Players")
local RunService          = game:GetService("RunService")
local TweenService        = game:GetService("TweenService")
local UserInputService    = game:GetService("UserInputService")
local VirtualInputManager = game:GetService("VirtualInputManager")

local LocalPlayer = Players.LocalPlayer
local Camera      = workspace.CurrentCamera or workspace:WaitForChild("Camera")
local Mouse       = LocalPlayer:GetMouse()

-- ═══════════════════════ CONFIG ═══════════════════════

local CONFIG = {
    Radius       = 250,    -- rayon de recherche des PNJ (studs)
    LockFov      = 180,    -- demi-angle de lock : 180 = vise tout autour (360°). Baisse pour restreindre à l'écran.
    NeedSight    = true,   -- ne lock que si ligne de vue dégagée (raycast)
    FireInterval = 0.075,  -- intervalle entre deux clics gauches (s) — 75 ms
    FireJitter   = 0.0,    -- aléatoire ajouté à l'intervalle (± s) — 0 = cadence pile
    ClickHold    = 0.02,   -- durée d'appui du clic gauche dans chaque cycle (s)
    LeadFactor   = 0.08,   -- anticipation sur la vélocité de la tête (s)
    AimJitter    = 0.05,   -- micro-jitter sur le point visé (± studs)
    LootInterval  = 0.3,   -- fréquence du scan de loot (s) — compte les drops à portée
    ESpamInterval = 0.2,   -- intervalle entre deux appuis sur E (spam ramassage)
    LootReach    = 0,      -- portée bonus au-delà du MaxActivationDistance
    LootKey      = Enum.KeyCode.E,
    WalkToLoot   = false,  -- true = marcher jusqu'au loot hors portée
    ScanInterval = 1.0,    -- intervalle du rescan des PNJ (s)
}

-- ═══════════════════════ ÉTAT ═══════════════════════

local State = {
    Enabled    = false,
    Alive      = true,
    Locked     = false,
    RightHeld  = false,
    LastTarget = nil,
    Kills      = 0,
}

local NpcCache  = {}   -- [Model] = true
local NpcCount  = 0
local LootCount = 0
local LocalChar = LocalPlayer.Character
local Cleanup   = {}

-- ═══════════════════════ HELPERS ═══════════════════════

local function BindCleanup(connection)
    table.insert(Cleanup, connection)
    return connection
end

local function NewInstance(class, parent, props)
    local obj = Instance.new(class)
    for key, value in pairs(props) do
        obj[key] = value
    end
    obj.Parent = parent
    return obj
end

local function AddCorner(parent, radius)
    return NewInstance("UICorner", parent, { CornerRadius = UDim.new(0, radius) })
end

-- ═══════════════════════ SCAN DES PNJ ═══════════════════════

local function IsPlayerCharacter(char)
    -- DOUBLE VÉRIFICATION — un joueur ne doit JAMAIS être ciblé.
    if Players:GetPlayerFromCharacter(char) ~= nil then
        return true
    end
    if char:FindFirstChild("Player") ~= nil then
        return true
    end
    return false
end

local function RescanNpcs()
    LocalChar = LocalPlayer.Character
    local found = {}
    for _, descendant in ipairs(workspace:GetDescendants()) do
        if descendant:IsA("Humanoid") then
            local humanoid = descendant
            if humanoid.Health > 0 then
                local char = humanoid.Parent
                if char and char:IsA("Model") and not found[char] then
                    local head = char:FindFirstChild("Head")
                    if head
                        and head:IsA("BasePart")
                        and not IsPlayerCharacter(char)
                        and char ~= LocalChar
                    then
                        found[char] = true
                    end
                end
            end
        end
    end
    NpcCache = found
    NpcCount = 0
    for _ in pairs(found) do
        NpcCount += 1
    end
end

-- ═══════════════════════ LIGNE DE VUE (raycast) ═══════════════════════
-- Même principe que le wall check de SilentAimNPC.lua : un raycast
-- caméra → tête. Si le premier obstacle n'est pas le PNJ lui-même,
-- il est derrière un mur → pas de lock.

local function IsBlocked(targetChar, aimPoint)
    local camPos = Camera.CFrame.Position
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local filter = {}
    if LocalChar then
        table.insert(filter, LocalChar)
    end
    params.FilterDescendantsInstances = filter

    local result = workspace:Raycast(camPos, aimPoint - camPos, params)
    if not result then
        return false
    end
    return not result.Instance:IsDescendantOf(targetChar)
end

-- ═══════════════════════ CIBLE ═══════════════════════
-- Le plus proche PNJ qui remplit TOUT :
--   • dans le rayon,
--   • dans le cône LockFov (180 = 360° complet, verrouille partout),
--   • ligne de vue dégagée (NeedSight) : pas de tir à travers les murs.

local function GetVisibleTarget()
    local camPos = Camera.CFrame.Position
    local look   = Camera.CFrame.LookVector
    local best, bestDist = nil, CONFIG.Radius

    for char in pairs(NpcCache) do
        local head = char:FindFirstChild("Head")
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if head and head:IsA("BasePart") and humanoid and humanoid.Health > 0 then
            local delta = head.Position - camPos
            local dist  = delta.Magnitude
            if dist <= CONFIG.Radius and dist < bestDist then

                -- FOV : le PNJ doit être visible à l'écran
                local angle = math.deg(math.acos(math.clamp(delta.Unit:Dot(look), -1, 1)))
                if angle <= CONFIG.LockFov then

                    -- LIGNE DE VUE : pas de mur entre caméra et tête
                    if not CONFIG.NeedSight or not IsBlocked(char, head.Position) then
                        best, bestDist = char, dist
                    end
                end
            end
        end
    end
    return best
end

-- ═══════════════════════ AIM (vise tout seul) ═══════════════════════
-- HOOK AIM — RenderStepped : dès qu'un PNJ est dans le rayon avec
-- ligne de vue, la caméra se tourne toute seule dessus (lead +
-- jitter) et le tir suit. Aucune cible → la caméra est rendue au
-- jeu (CameraType = Custom).

local function JitterVector(range)
    return Vector3.new(
        (math.random() * 2 - 1) * range,
        (math.random() * 2 - 1) * range,
        (math.random() * 2 - 1) * range
    )
end

BindCleanup(RunService.RenderStepped:Connect(function()
    if not State.Enabled then
        return
    end
    local target = GetVisibleTarget()
    if not target then
        State.Locked = false
        Camera.CameraType = Enum.CameraType.Custom
        return
    end
    local head = target:FindFirstChild("Head")
    if not head then
        State.Locked = false
        Camera.CameraType = Enum.CameraType.Custom
        return
    end
    Camera.CameraType = Enum.CameraType.Scriptable
    local aimPoint = head.Position
        + head.AssemblyLinearVelocity * CONFIG.LeadFactor
        + JitterVector(CONFIG.AimJitter)
    Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, aimPoint)
    State.Locked = true
    State.LastTarget = target
end))

-- ═══════════════════════ FIRE (tout automatique) ══════════════
-- HOOK FIRE — dès que FARM est ON, AUCUNE touche à presser :
--   • clic droit MAINTENU (viser / ADS) — enfoncé une fois, jamais relâché
--     tant que le farm tourne,
--   • clic gauche toutes les 75 ms (mouse1down/mouse1up en boucle),
--   • Tool:Activate() sur l'arme équipée à chaque cycle.
-- Les trois chemins partent tout seuls : le jeu répond à l'un ou
-- l'autre, le tir sort.

local function PressRight()
    -- clic droit MAINTENU : on l'enfonce une seule fois puis on le laisse
    -- enfoncé. State.RightHeld empêche de spammer Button2Down à chaque
    -- cycle (un jeu en mode toggle basculerait l'ADS à chaque appui).
    if State.RightHeld then
        return
    end
    State.RightHeld = true
    pcall(function()
        Mouse:Button2Down()
    end)
end

local function ReleaseFire()
    State.RightHeld = false
    pcall(function()
        Mouse:Button2Up()
    end)
    pcall(function()
        Mouse:Button1Up()
    end)
end

-- BOUCLE DE TIR — clic droit maintenu + clic gauche toutes les 75 ms.
-- Un cycle = appui gauche (ClickHold) puis relâchement, puis attente du
-- reste pour que la période totale tombe pile sur FireInterval.
task.spawn(function()
    while State.Alive do
        if State.Enabled then
            PressRight()

            pcall(function()
                Mouse:Button1Down()
            end)
            -- arme équipée : activation directe du Tool
            local char = LocalChar
            local tool = char and char:FindFirstChildOfClass("Tool")
            if tool then
                pcall(function()
                    tool:Activate()
                end)
            end
            task.wait(CONFIG.ClickHold)
            pcall(function()
                Mouse:Button1Up()
            end)

            local target = State.LastTarget
            if target then
                local humanoid = target:FindFirstChildOfClass("Humanoid")
                if humanoid and humanoid.Health <= 0 then
                    State.Kills += 1
                    State.LastTarget = nil
                end
            end

            local cycle = CONFIG.FireInterval + math.random() * CONFIG.FireJitter
            task.wait(math.max(cycle - CONFIG.ClickHold, 0))
        else
            task.wait(CONFIG.FireInterval)
        end
    end
end)

-- ═══════════════════════ LOOT (touche E, sans bouger) ═══════════════════
-- HOOK LOOT — deux étages :
--   1. TryLoot compte les ProximityPrompt à portée (compteur Loots).
--   2. La boucle de SPAM appuie sur E en continu via VirtualInputManager :
--      tout ce qui est ramassable à portée finit dans la poche.

local function GetRootPosition()
    if LocalChar then
        local root = LocalChar:FindFirstChild("HumanoidRootPart")
        if root then
            return root.Position
        end
    end
    return Camera.CFrame.Position
end

local function TryLoot()
    local origin = GetRootPosition()
    local inRange = {}

    for _, descendant in ipairs(workspace:GetDescendants()) do
        if descendant:IsA("ProximityPrompt") and descendant.Enabled then
            local holder = descendant.Parent
            if holder and holder:IsA("BasePart") then
                local dist = (holder.Position - origin).Magnitude
                if dist <= descendant.MaxActivationDistance + CONFIG.LootReach then
                    table.insert(inRange, descendant)
                end
            end
        end
    end

    LootCount = #inRange

    -- L'appui sur E est maintenant fait par la boucle de SPAM :
    -- TryLoot ne sert plus qu'à compter les drops à portée.

    -- Optionnel : marcher jusqu'au loot hors portée (WalkToLoot = true).
    if CONFIG.WalkToLoot then
        local root = LocalChar and LocalChar:FindFirstChild("HumanoidRootPart")
        local humanoid = LocalChar and LocalChar:FindFirstChildOfClass("Humanoid")
        if root and humanoid then
            local best, bestDist = nil, 80
            for _, descendant in ipairs(workspace:GetDescendants()) do
                if descendant:IsA("ProximityPrompt") and descendant.Enabled then
                    local holder = descendant.Parent
                    if holder and holder:IsA("BasePart") then
                        local dist = (holder.Position - origin).Magnitude
                        if dist < bestDist then
                            best, bestDist = holder, dist
                        end
                    end
                end
            end
            if best then
                humanoid:MoveTo(best.Position)
            end
        end
    end
end

task.spawn(function()
    while State.Alive do
        if State.Enabled then
            pcall(TryLoot)
            -- SPAM E : appuie en continu pour ramasser tout ce qui est
            -- à portée, même si le scan ProximityPrompt ne le voit pas
            -- (loot custom, prompts désactivés, etc.).
            VirtualInputManager:SendKeyEvent(true, CONFIG.LootKey, false, game)
            task.wait(0.05)
            VirtualInputManager:SendKeyEvent(false, CONFIG.LootKey, false, game)
        end
        task.wait(CONFIG.ESpamInterval)
    end
end)

-- ═══════════════════════ FERMETURE ═══════════════════════

local function CloseAll()
    State.Alive = false
    State.Enabled = false
    ReleaseFire()
    pcall(function()
        Camera.CameraType = Enum.CameraType.Custom
    end)
    for _, connection in ipairs(Cleanup) do
        pcall(function()
            connection:Disconnect()
        end)
    end
    ScreenGui:Destroy()
end

-- ═══════════════════════ LOOPS (stats + scan) ═══════════════════════

local Frames = 0
BindCleanup(RunService.RenderStepped:Connect(function()
    Frames += 1
end))

task.spawn(function()
    while State.Alive do
        task.wait(0.5)
        local fps = math.floor(Frames / 0.5 + 0.5)
        Frames = 0
        StatsLabel.Text = string.format(
            "NPCs: %d  |  Loots: %d  |  Kills: %d  |  FPS: %d  |  LOCK: %s",
            NpcCount, LootCount, State.Kills, fps,
            State.Locked and "ON" or "OFF"
        )
    end
end)

task.spawn(function()
    RescanNpcs()
    while State.Alive do
        task.wait(CONFIG.ScanInterval)
        RescanNpcs()
    end
end)

-- ═══════════════════════════════════════════════════════════════════
-- [RÉGLAGES PAR JEU — Urban Reload]
--   • LockFov : demi-angle du cône de lock. 180 = vise et tire à
--     360° autour de toi (défaut). Baisse à 60 pour un aimbot collant
--     mais limité à l'écran, 25 pour un lock très discret.
--   • NeedSight = false si le jeu n'a pas de vrais murs (lock FOV seul).
--   • Si les PNJ mettent du temps à mourir : monte FireInterval
--     (0.075 par défaut) pour laisser l'arme respirer entre deux clics.
--     Si le loot se ramasse au clic et pas à la touche : remplace le
--     scan ProximityPrompt par un scan de parts nommées et utilise
--     Mouse:Button1Down() visé sur le drop.
--   • Si le loot doit être ramassé en marchant : WalkToLoot = true.
--   • FireInterval : cadence du spam de clic gauche (0.075 par défaut = 75 ms).
--     Baisse à 0.05 pour tirer plus vite, monte si l'arme s'étouffe.
--   • ClickHold : durée d'appui de chaque clic gauche (0.02 par défaut).
--     Monte à 0.04 si le jeu ignore les clics trop courts.
--   • Le clic droit n'est enfoncé qu'une fois par activation (anti-toggle).
-- ═══════════════════════════════════════════════════════════════════
